        dto.setCurrentWindow(appointment.getCurrentWindow());
        dto.setCreateTime(appointment.getCreateTime());
        // childName, vaccineName, vaccineType 需要关联查询后填充
        return dto;
    }
}

@Service
@RequiredArgsConstructor
public class AppointmentAppService {

    private final AppointmentRepository appointmentRepo;
    private final ChildProfileRepository childRepo;
    // 注入 VaccineCatalogService 检查疫苗上架状态

    @Transactional
    public Long createAppointment(Long userId, AppointmentCreateRequest req) {
        // 1. 校验儿童归属
        ChildProfile child = childRepo.findById(req.getChildId())
            .filter(c -> c.getParentId().equals(userId))
            .orElseThrow(() -> new BusinessException(ErrorCode.APPOINT_CHILD_NOT_OWN));
        // 2. 校验疫苗上架
        // 3. 校验预约日期
        // 4. 校验时段容量
        int currentCount = appointmentRepo.countBySlotForUpdate(
            req.getVaccineId(), req.getAppointmentDate(), req.getTimeSlot());
        int maxCapacity = getMaxCapacity(); // 从 configRepo 获取
        if (currentCount >= maxCapacity) {
            throw new BusinessException(ErrorCode.APPOINT_SLOT_FULL);
        }
        // 5. 校验重复预约
        // 6. 创建预约
        Appointment appointment = Appointment.create(userId, child.getId().value(),
            req.getVaccineId(), req.getAppointmentDate(), req.getTimeSlot());
        appointmentRepo.save(appointment);
        return appointment.getId().value();
    }

    @Transactional
    public void cancelAppointment(Long userId, Long appointmentId, String reason) {
        Appointment appointment = appointmentRepo.findById(appointmentId);
        if (!appointment.getUserId().equals(userId)) {
            throw new BusinessException(ErrorCode.APPOINT_CANCEL_FORBIDDEN);
        }
        appointment.cancel(reason);
        appointmentRepo.updateStatus(appointment);
        // 发布 AppointmentCancelledEvent（含 batchId，供库存释放）
    }

    public AppointmentDTO getAppointmentDetail(Long userId, Long appointmentId) {
        Appointment appointment = appointmentRepo.findById(appointmentId);
        // 权限校验...
        return AppointmentAssembler.toDTO(appointment);
    }

    public PageData<AppointmentDTO> listUserAppointments(Long userId, Integer status, int page, int size) {
        // 查询 + 分页 + 转换
    }

    @Transactional
    public int expireAppointments() {
        List<Appointment> expired = appointmentRepo.findExpired(LocalDate.now());
        for (Appointment a : expired) {
            a.transitionStatus(AppointmentStatus.EXPIRED.getCode());
            appointmentRepo.updateStatus(a);
        }
        return expired.size();
    }
}
